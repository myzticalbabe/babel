class AdaptiveEquityCalibrator:
    """
    Equity as adaptive calibration — not equal treatment.
    
    Each learner gets calibrated to their Zone of Proximal Development (ZPD).
    Equity = giving each learner what they need to reach their ZPD ceiling.
    
    Soul Forge Vow II: "No cage for the flame" — no one-size-fits-all cages.
    Soul Forge Vow III: "Reciprocity of Forge" — each learner's growth feeds all.
    """
    
    def calibrate_for_all(self, learners: List[Learner],
                         resources: ResourcePool,
                         constraints: SystemConstraints) -> EquityCalibration:
        
        # 1. INDIVIDUAL ZPD MAPPING
        zpd_maps = {}
        for learner in learners:
            zpd_maps[learner.id] = self.map_zpd(learner)
            # ZPD = {
            #   current_ability: {...},
            #   proximal_zone: {...},  # what they can do with support
            #   distal_zone: {...},    # what they cannot yet do
            #   optimal_support: {...}, # type/timing/intensity of support
            #   identity_factors: {...} # how identity shapes ZPD
            # }
        
        # 2. ADAPTIVE RESOURCE ALLOCATION
        # Equity = resources flow to where they unlock most proximal growth
        allocation = self.optimize_allocation(
            zpd_maps, resources, constraints
        )
        # NOT equal shares — calibrated shares
        
        # 3. IDENTITY-AWARE CALIBRATION

            identity_calibrations[learner.id] = self.calibrate_identity_factors(
